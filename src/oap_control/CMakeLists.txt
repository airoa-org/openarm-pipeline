cmake_minimum_required(VERSION 3.8)
project(oap_control)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -Wpedantic)
set(DEPS ament_index_cpp controller_interface hardware_interface pluginlib rclcpp rclcpp_lifecycle realtime_tools
         geometry_msgs sensor_msgs std_msgs std_srvs trajectory_msgs)
find_package(ament_cmake REQUIRED)
foreach(d ${DEPS})
  find_package(${d} REQUIRED)
endforeach()
find_package(pinocchio REQUIRED)  # the source build, through CMAKE_PREFIX_PATH (docker/Dockerfile)
add_library(oap_control SHARED src/model.cpp src/spline.cpp src/ik.cpp src/motion.cpp src/controller.cpp)
target_include_directories(oap_control PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)
target_link_libraries(oap_control pinocchio::pinocchio)
ament_target_dependencies(oap_control ${DEPS})
pluginlib_export_plugin_description_file(controller_interface plugins.xml)
install(TARGETS oap_control LIBRARY DESTINATION lib)
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  foreach(t test_spline test_inputs test_ik)  # test_ik reads oap_description's URDF, through ament_index
    ament_add_gtest(${t} test/${t}.cpp)
    target_link_libraries(${t} oap_control)
    ament_target_dependencies(${t} geometry_msgs trajectory_msgs)
  endforeach()
  target_compile_definitions(test_ik PRIVATE OAP_CONTROLLERS_YAML="${CMAKE_CURRENT_SOURCE_DIR}/config/controllers.yaml")
endif()
ament_package()
